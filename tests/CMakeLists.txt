cmake_minimum_required(VERSION 3.20)
project(FireteamPortableTests LANGUAGES CXX)
enable_testing()
# Intentionally independent of the proprietary Jupiter SDK and game CMake.
file(GLOB FT_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test-*.cpp")
foreach(source IN LISTS FT_TEST_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_compile_features(${name} PRIVATE cxx_std_11)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endforeach()
